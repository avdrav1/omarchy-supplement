# Apps installed through Omarchy's own installers (`omarchy install ...`),
# which add Omarchy theming and defaults on top of the package. Each line is:
#   <command that exists once installed>   <omarchy install arguments>
# The installer is skipped when the command already exists. Note that these
# installers open the app when they finish.

ghostty     terminal ghostty    # Ghostty as the default terminal (SUPER+Return)
code        editor vscode       # VS Code + libsecret password store + theme
zeditor     editor zed          # Zed + omazed theme sync
slack       service slack       # Slack with live Omarchy theme
1password   service 1password   # 1Password + CLI + browser extension
steam       gaming steam        # lib32 GPU drivers first, then Steam
spotify     service spotify     # the app on x86, the web player on ARM
herdr-app   ai herdr            # Herdr desktop app (updates itself) + Omarchy keys and theme
